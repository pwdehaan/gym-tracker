function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Logs var logSheet = ss.getSheetByName("Logs"); var logs = []; if (logSheet) { var logData = logSheet.getDataRange().getValues(); for (var i = 1; i < logData.length; i++) { if (!logData[i][0]) continue; logs.push({ id: String(logData[i][0]), workoutId: String(logData[i][1]), date: String(logData[i][2]), startTime: String(logData[i][3]), exerciseId: String(logData[i][4]), exerciseName: String(logData[i][5]), setNum: Number(logData[i][6]), primaryValue: String(logData[i][7]), weightValue: String(logData[i][8]), extraValue: String(logData[i][9]), notes: String(logData[i][10]), work_kj: Number(logData[i][11] || 0), energy_kcal: Number(logData[i][12] || 0) }); } } // 2. Oefeningen var exSheet = ss.getSheetByName("Exercises"); var exercises = []; if (exSheet) { var exData = exSheet.getDataRange().getValues(); for (var j = 1; j < exData.length; j++) { if (!exData[j][0]) continue; exercises.push({ id: String(exData[j][0]), name: String(exData[j][1]), volume: String(exData[j][2]), weight: String(exData[j][3]), isHyrox: String(exData[j][4]) === "true", isStation: String(exData[j][5]) === "true", target: String(exData[j][6] || ''), notes: String(exData[j][7] || ''), icon: String(exData[j][8] || '🏋️'), formula: String(exData[j][9] || '') }); } } // 3. Profielen (Fysiologie) var profSheet = ss.getSheetByName("Profiles"); var profiles = []; if (profSheet) { var profData = profSheet.getDataRange().getValues(); for (var k = 1; k < profData.length; k++) { if (!profData[k][0]) continue; profiles.push({ id: String(profData[k][0]), date: String(profData[k][1]), height: Number(profData[k][2]), weight: Number(profData[k][3]), efficiency: Number(profData[k][4]) }); } } return ContentService.createTextOutput(JSON.stringify({ status: "success", logs: logs, exercises: exercises, profiles: profiles })) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { var contents = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); // --- LOGS --- if (contents.logs) { var logSheet = ss.getSheetByName("Logs") || ss.insertSheet("Logs"); if (logSheet.getLastRow() === 0) { logSheet.appendRow(["id", "workoutId", "date", "startTime", "exerciseId", "exerciseName", "setNum", "primaryValue", "weightValue", "extraValue", "notes", "work_kj", "energy_kcal"]); } var existingLogs = logSheet.getDataRange().getValues(); var logMap = {}; for (var i = 1; i < existingLogs.length; i++) { if (existingLogs[i][0]) logMap[String(existingLogs[i][0])] = existingLogs[i]; } contents.logs.forEach(function(l) { logMap[String(l.id)] = [ String(l.id), String(l.workoutId||''), String(l.date||''), String(l.startTime||''), String(l.exerciseId||''), String(l.exerciseName||''), Number(l.setNum||1), String(l.primaryValue||''), String(l.weightValue||''), String(l.extraValue||''), String(l.notes||''), Number(l.work_kj||0), Number(l.energy_kcal||0) ]; }); logSheet.clearContents(); logSheet.appendRow(["id", "workoutId", "date", "startTime", "exerciseId", "exerciseName", "setNum", "primaryValue", "weightValue", "extraValue", "notes", "work_kj", "energy_kcal"]); var logRows = Object.keys(logMap).map(function(k) { return logMap[k]; }); if (logRows.length > 0) logSheet.getRange(2, 1, logRows.length, 13).setValues(logRows); } // --- EXERCISES --- if (contents.exercises) { var exSheet = ss.getSheetByName("Exercises") || ss.insertSheet("Exercises"); if (exSheet.getLastRow() === 0) { exSheet.appendRow(["id", "name", "volume", "weight", "isHyrox", "isStation", "target", "notes", "icon", "formula"]); } var existingEx = exSheet.getDataRange().getValues(); var exMap = {}; for (var j = 1; j < existingEx.length; j++) { if (existingEx[j][0]) exMap[String(existingEx[j][0])] = existingEx[j]; } contents.exercises.forEach(function(x) { exMap[String(x.id)] = [ String(x.id), String(x.name||''), String(x.volume||''), String(x.weight||''), x.isHyrox?"true":"false", x.isStation?"true":"false", String(x.target||''), String(x.notes||''), String(x.icon||'🏋️'), String(x.formula||'') ]; }); exSheet.clearContents(); exSheet.appendRow(["id", "name", "volume", "weight", "isHyrox", "isStation", "target", "notes", "icon", "formula"]); var exRows = Object.keys(exMap).map(function(k) { return exMap[k]; }); if (exRows.length > 0) exSheet.getRange(2, 1, exRows.length, 10).setValues(exRows); } // --- PROFILES --- if (contents.profiles) { var profSheet = ss.getSheetByName("Profiles") || ss.insertSheet("Profiles"); if (profSheet.getLastRow() === 0) { profSheet.appendRow(["id", "date", "height", "weight", "efficiency"]); } var existingProf = profSheet.getDataRange().getValues(); var profMap = {}; for (var k = 1; k < existingProf.length; k++) { if (existingProf[k][0]) profMap[String(existingProf[k][0])] = existingProf[k]; } contents.profiles.forEach(function(p) { profMap[String(p.id)] = [ String(p.id), String(p.date||''), Number(p.height||1.97), Number(p.weight||87), Number(p.efficiency||0.23) ]; }); profSheet.clearContents(); profSheet.appendRow(["id", "date", "height", "weight", "efficiency"]); var profRows = Object.keys(profMap).map(function(k) { return profMap[k]; }); if (profRows.length > 0) profSheet.getRange(2, 1, profRows.length, 5).setValues(profRows); } return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON); } }
